# Pin every plugin EXACTLY and resolve them together in one `uv pip install`.
# Do not install these one-by-one or in separate RUN layers — that lets pip/uv
# resolve each plugin's dependencies in isolation, which is what silently
# breaks NetBox when a second or third plugin is added.
#
# Re-run `docker compose build --no-cache` after every change to this file.

# Pinned to NetBox 4.6.x on purpose: netbox-secrets 3.1.1 declares
# max_version 4.6.99 and hasn't shipped a 4.7-compatible release yet, and
# netbox-bgp declares an exact min/max per release (0.20.1 -> 4.7 only,
# 0.19.0 -> 4.6 only). This is the newest combination where both plugins
# actually load — verified by building the image and checking for NetBox's
# "Unable to load plugin" warnings in `docker compose build` output.
netbox-secrets==3.1.1
netbox-bgp==0.19.0

# Alpha-quality (pre-1.0). Pin exact versions, read the changelog before any
# bump, and back up the DB before running migrations after an upgrade.
# Requires NetBox 4.5+, so it's fine on 4.6.
netbox-fms==0.4.1

# Hardware inventory tracking. min_version 4.6, no max declared.
netbox-inventory==2.6.1

# TOTP two-factor login. Supports NetBox 4.0-4.99.
# IMPORTANT: otp_required defaults to True upstream, which locks every user
# out at next login until they register a device via `manage.py addtotp`.
# We override it to False below until you're ready to enforce it.
netbox-otp-plugin==1.3.6

# Deliberately pinned one minor behind latest (0.10.0): that release requires
# NetBox 4.7 only. 0.9.2 is the last one that supports our NetBox 4.6 pin.
netbox-floorplan-plugin==0.9.2

# Alpha software (data model/API may still change). min_version 4.5, no max.
# Optionally integrates with netbox-pathways if that's ever added (soft
# import, not required).
netbox-wdm==0.2.2
